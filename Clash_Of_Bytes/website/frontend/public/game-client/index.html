TASK 2: TRANSFORMATIONS & CLIPPING CO5: LINE CLIPPING ALGORITHMS [L3] PRACTICAL 05

Cohen–Sutherland 2D Line Clipping Algorithm

Textbook 4-bit region outcodes, bitwise trivial accept/reject, and parametric boundary intersection computation against viewport bounds.

INITIALIZED
Mathematical Principle & Bitwise Outcode Mapping:
Viewport defined by boundaries: [Xmin, Ymin, Xmax, Ymax]. Each endpoint is assigned a 4-bit region code [TOP | BOTTOM | RIGHT | LEFT]:
• INSIDE (0000 = 0): Point lies within viewport.
• LEFT (0001 = 1): x < Xmin • RIGHT (0010 = 2): x > Xmax
• BOTTOM (0100 = 4): y > Ymax • TOP (1000 = 8): y < Ymin
• Trivial Accept: (code0 | code1) == 0000 → Entire line segment is inside and visible.
• Trivial Reject: (code0 & code1) != 0000 → Both points share an exterior zone; line lies completely outside.
• Intersection Equations: For crossing lines: x = x0 + (x1 - x0) * (y_boundary - y0) / (y1 - y0) and y = y0 + (y1 - y0) * (x_boundary - x0) / (x1 - x0).
• Clash of Bytes Application: In GameMatch.cpp line 228, actively clips dynamic laser beams against combat viewport [280, 160, 1000, 520].

#ifndef CO5_PRAC05_COHEN_SUTHERLAND_CLIPPING_HPP
#define CO5_PRAC05_COHEN_SUTHERLAND_CLIPPING_HPP

#include <iostream>
#include <iomanip>

/*
 * Subject: Computer Graphics
 * Course Outcome: CO5 | Practical 05
 * Topic: Cohen-Sutherland 2D Line Clipping Algorithm
 * Purpose: Clips a 2D line segment against a rectangular viewport using 4-bit region outcodes.
 */

class CohenSutherlandClipping {
public:
    // 4-bit region outcodes
    static const int INSIDE = 0; // 0000
    static const int LEFT   = 1; // 0001
    static const int RIGHT  = 2; // 0010
    static const int BOTTOM = 4; // 0100
    static const int TOP    = 8; // 1000

    // Compute 4-bit region code for point (x, y) relative to clipping window [xmin, ymin, xmax, ymax]
    static int computeOutCode(double x, double y, double xmin, double ymin, double xmax, double ymax) {
        int code = INSIDE;

        if (x < xmin) {
            code |= LEFT;
        } else if (x > xmax) {
            code |= RIGHT;
        }

        if (y < ymin) {
            code |= BOTTOM;
        } else if (y > ymax) {
            code |= TOP;
        }

        return code;
    }

    // Cohen-Sutherland line clipping algorithm
    // Clips line segment from (x0, y0) to (x1, y1) against clipping window
    // Returns true if any portion of the line is visible (accepted), false if completely outside (rejected)
    static bool clipLine(double& x0, double& y0, double& x1, double& y1,
                         double xmin, double ymin, double xmax, double ymax) {
        int code0 = computeOutCode(x0, y0, xmin, ymin, xmax, ymax);
        int code1 = computeOutCode(x1, y1, xmin, ymin, xmax, ymax);
        bool accept = false;

        while (true) {
            if ((code0 | code1) == 0) {
                // Trivial accept: both endpoints are inside viewport
                accept = true;
                break;
            } else if ((code0 & code1) != 0) {
                // Trivial reject: both endpoints share an outside zone
                break;
            } else {
                // Line crosses the clipping boundary; calculate intersection point
                double x = 0.0, y = 0.0;
                // Pick an endpoint that is outside the clipping window
                int codeOut = code0 ? code0 : code1;

                if (codeOut & TOP) {
                    // Point is above the clipping window
                    x = x0 + (x1 - x0) * (ymax - y0) / (y1 - y0);
                    y = ymax;
                } else if (codeOut & BOTTOM) {
                    // Point is below the clipping window
                    x = x0 + (x1 - x0) * (ymin - y0) / (y1 - y0);
                    y = ymin;
                } else if (codeOut & RIGHT) {
                    // Point is to the right of clipping window
                    y = y0 + (y1 - y0) * (xmax - x0) / (x1 - x0);
                    x = xmax;
                } else if (codeOut & LEFT) {
                    // Point is to the left of clipping window
                    y = y0 + (y1 - y0) * (xmin - x0) / (x1 - x0);
                    x = xmin;
                }

                // Update the outside endpoint coordinates and recompute its outcode
                if (codeOut == code0) {
                    x0 = x;
                    y0 = y;
                    code0 = computeOutCode(x0, y0, xmin, ymin, xmax, ymax);
                } else {
                    x1 = x;
                    y1 = y;
                    code1 = computeOutCode(x1, y1, xmin, ymin, xmax, ymax);
                }
            }
        }

        return accept;
    }

    // Standalone unit test verifying algorithm correctness with a line crossing the viewport
    static void testClipping() {
        std::cout << "\n=======================================================\n";
        std::cout << "  COMPUTER GRAPHICS - PRACTICAL 05: COHEN-SUTHERLAND\n";
        std::cout << "  Line Clipping Algorithm Test Verification\n";
        std::cout << "=======================================================\n";

        double xmin = 100.0, ymin = 100.0, xmax = 400.0, ymax = 300.0;
        std::cout << "Clipping Viewport: [" << xmin << ", " << ymin << "] to [" << xmax << ", " << ymax << "]\n\n";

        // Test 1: Line crossing viewport
        double x0 = 50.0, y0 = 50.0, x1 = 450.0, y1 = 350.0;
        std::cout << "Test 1 - Line Crossing Viewport:\n";
        std::cout << "  Original Endpoints: (" << x0 << ", " << y0 << ") -> (" << x1 << ", " << y1 << ")\n";

        bool visible = clipLine(x0, y0, x1, y1, xmin, ymin, xmax, ymax);
        std::cout << "  Clipping Result   : " << (visible ? "ACCEPTED (Visible)" : "REJECTED (Invisible)") << "\n";
        if (visible) {
            std::cout << "  Clipped Endpoints : (" << std::fixed << std::setprecision(1) << x0 
                      << ", " << y0 << ") -> (" << x1 << ", " << y1 << ")\n";
        }

        // Test 2: Completely outside line
        double ox0 = 10.0, oy0 = 10.0, ox1 = 50.0, oy1 = 80.0;
        std::cout << "\nTest 2 - Line Completely Outside (Trivial Reject):\n";
        std::cout << "  Original Endpoints: (" << ox0 << ", " << oy0 << ") -> (" << ox1 << ", " << oy1 << ")\n";
        bool outVisible = clipLine(ox0, oy0, ox1, oy1, xmin, ymin, xmax, ymax);
        std::cout << "  Clipping Result   : " << (outVisible ? "ACCEPTED" : "TRIVIAL REJECT") << "\n";

        std::cout << "=======================================================\n\n";
    }
};

Bresenham's Integer Line Drawing Algorithm

Precise integer decision variable line rasterization without floating-point division or trigonometry.

Decision Variable Mathematical Formula:
For slope 0 ≤ m ≤ 1: Initial decision parameter d0 = 2Δy - Δx.
At step k: If dk < 0, select pixel (xk + 1, yk) and d(k+1) = dk + 2Δy.
If dk ≥ 0, select pixel (xk + 1, yk + 1) and d(k+1) = dk + 2Δy - 2Δx.

/*
 * Subject: Computer Graphics
 * Course Outcome: CO3 | Practical 02
 * Topic: Bresenham's Line & 8-Way Symmetric Circle Drawing Algorithm
 * In-Game Application: Dynamic hitsparks, slash rays, and circular shield barriers
 * operating entirely on integer decision parameters without trigonometric functions!
 */
class BresenhamVFX {
public:
    // Integer-based Bresenham Line Drawing
    static void drawLine(int x1, int y1, int x2, int y2, float r, float g, float b) {
        int dx = std::abs(x2 - x1);
        int dy = std::abs(y2 - y1);
        int sx = (x1 < x2) ? 1 : -1;
        int sy = (y1 < y2) ? 1 : -1;
        int err = dx - dy;

        glColor3f(r, g, b);
        glBegin(GL_POINTS);
        while (true) {
            glVertex2i(x1, y1);
            if (x1 == x2 && y1 == y2) break;
            int e2 = 2 * err;
            if (e2 > -dy) {
                err -= dy;
                x1 += sx;
            }
            if (e2 < dx) {
                err += dx;
                y1 += sy;
            }
        }
        glEnd();
    }

    // Integer-based 8-Way Symmetric Bresenham Circle Drawing
    static void drawCircle(int xc, int yc, int radius, float r, float g, float b) {
        int x = 0;
        int y = radius;
        int d = 3 - 2 * radius; // Initial decision parameter

        glColor3f(r, g, b);
        glBegin(GL_POINTS);
        while (y >= x) {
            // Plot all 8 octants
            glVertex2i(xc + x, yc + y);
            glVertex2i(xc - x, yc + y);
            glVertex2i(xc + x, yc - y);
            glVertex2i(xc - x, yc - y);
            glVertex2i(xc + y, yc + x);
            glVertex2i(xc - y, yc + x);
            glVertex2i(xc + y, yc - x);
            glVertex2i(xc - y, yc - x);

            if (d > 0) {
                y--;
                d = d + 4 * (x - y) + 10;
            } else {
                d = d + 4 * x + 6;
            }
            x++;
        }
        glEnd();
    }
};

TASK 4: RENDERING THE APPLICATION CO3 & CO4: POLYGON RASTERIZATION [L3] PRACTICAL 03

Scan-Line Polygon Filling Algorithm & Health Bars

Horizontal scanline sweep, edge-table intersection calculation, ascending sort, and parity-rule pixel span filling.

Edge Intersection & Parity Checking:
For each horizontal scanline y from minY to maxY:
1. Find edges intersecting y where (y1 ≤ y < y2) || (y2 ≤ y < y1).
2. Calculate intersection X coordinate: x = x1 + (y - y1) / (y2 - y1) * (x2 - x1).
3. Sort intersection points in ascending order: [x0, x1, x2, x3, ...].
4. Fill horizontal pixel spans between pairs: [x0 .. x1], [x2 .. x3] (Parity rule).
• In-Game Application: Renders live player health bars in CO3_Prac03_ScanLine_Polygon_HealthBar.hpp and GameMatch.cpp!

#ifndef CO3_PRAC03_SCANLINE_POLYGON_HEALTHBAR_HPP
#define CO3_PRAC03_SCANLINE_POLYGON_HEALTHBAR_HPP

#include <vector>
#include <algorithm>
#include <GL/gl.h>

/*
 * Subject: Computer Graphics
 * Course Outcome: CO3 | Practical 03
 * Topic: Scan-Line Polygon Filling Algorithm with Parity Checking
 * In-Game Application: Rendering dynamic player health bar gauges line by line
 * with edge-intersection sorting and span filling!
 */
class ScanLinePolygonFill {
public:
    static void fillPolygon(const std::vector<std::pair<int, int>>& vertices, float r, float g, float b) {
        if (vertices.size() < 3) return;

        int minY = vertices[0].second;
        int maxY = vertices[0].second;
        for (const auto& v : vertices) {
            if (v.second < minY) minY = v.second;
            if (v.second > maxY) maxY = v.second;
        }

        glColor3f(r, g, b);
        glBegin(GL_LINES);

        // Cast scanline from minY to maxY
        for (int y = minY; y <= maxY; y++) {
            std::vector<int> intersections;
            size_t n = vertices.size();

            for (size_t i = 0; i < n; i++) {
                size_t j = (i + 1) % n;
                int y1 = vertices[i].second;
                int y2 = vertices[j].second;
                int x1 = vertices[i].first;
                int x2 = vertices[j].first;

                if ((y1 <= y && y2 > y) || (y2 <= y && y1 > y)) {
                    // Compute intersection point X
                    float x = x1 + (float)(y - y1) / (float)(y2 - y1) * (x2 - x1);
                    intersections.push_back((int)x);
                }
            }

            // Sort intersection points
            std::sort(intersections.begin(), intersections.end());

            // Fill pixel spans between pairs of intersections (Parity checking)
            for (size_t k = 0; k + 1 < intersections.size(); k += 2) {
                glVertex2i(intersections[k], y);
                glVertex2i(intersections[k + 1], y);
            }
        }

        glEnd();
    }
};

 * 2. ILLUMINATION & LIGHTING PIPELINE (PHONG / GOURAUD ILLUMINATION):

    glLoadIdentity();
    if (screenShakeTimer > 0.0f) {
        float shakeX = ((rand() % 100) / 50.0f - 1.0f) * screenShakeIntensity;
        float shakeY = ((rand() % 100) / 50.0f - 1.0f) * screenShakeIntensity;
        glTranslatef(shakeX, shakeY, 0.0f);
    }
}

void Renderer::update(float dt) {
    if (screenShakeTimer > 0.0f) {
        screenShakeTimer -= dt;
        if (screenShakeTimer <= 0.0f) {
            screenShakeTimer = 0.0f;
            screenShakeIntensity = 0.0f;
        }
    }
    cubeSpinAngle += 60.0f * dt;
    if (cubeSpinAngle >= 360.0f) cubeSpinAngle -= 360.0f;

    // Practical 06 (CO6): Update dynamic combat point light decay
    lightingEngine.update(dt);
}

AABB Overlap Condition:
Two boxes Box1[x1, y1, w1, h1] and Box2[x2, y2, w2, h2] intersect if and only if:
(x1 < x2 + w2) && (x1 + w1 > x2) && (y1 < y2 + h2) && (y1 + h1 > y2).

#include <string>

class CollisionSystem {
private:
    std::string latestALUTrace;

    static bool checkAABB(const Hitbox& a, const Hitbox& b) {
        return (a.offsetX < b.offsetX + b.width &&
                a.offsetX + a.width > b.offsetX &&
                a.offsetY < b.offsetY + b.height &&
                a.offsetY + a.height > b.offsetY);
    }

public:
    CollisionSystem() : latestALUTrace("WAITING FOR COMBAT HIT...") {}

    void checkCombat(Fighter* p1, Fighter* p2, ParticleSystem& particles, Renderer& renderer) {
        if (!p1 || !p2) return;
        if (!p1->isAlive() || !p2->isAlive()) return;

        // 1. Player Body Pushback
        float dist = p2->getPosition().x - p1->getPosition().x;
        float minDist = 70.0f;
        if (std::abs(dist) < minDist) {
            float pushDir = (dist >= 0.0f) ? 1.0f : -1.0f;
            // Push apart
            p1->takeHit(0, -pushDir * 0.2f, false);
            p2->takeHit(0, pushDir * 0.2f, false);
        }

        // Keep fighters facing each other
        if (p1->getPosition().x < p2->getPosition().x) {
            p1->setFacingRight(true);
            p2->setFacingRight(false);
        } else {
            p1->setFacingRight(false);
            p2->setFacingRight(true);
        }

        // 2. Check P1 attacking P2
        Hitbox p1Hit = p1->getActiveHitboxWorld();
        Hitbox p2Hurt = p2->getHurtboxWorld();
        if (p1Hit.active && checkAABB(p1Hit, p2Hurt)) {
            // Practical No. 03: REAL 64-BIT ASSEMBLY DAMAGE CALCULATION!
            CombatALUResult alu = AssemblyDamageEngine::calculateHitDamage(p1->getAttackDamage(), 5, 1);
            latestALUTrace = "P1 HIT! " + alu.assemblyTrace;

            float knockDir = p1->isFacingRightDir() ? 1.0f : -1.0f;
            p2->takeHit((int)alu.finalDamage, knockDir, p1->getState() == FighterState::SPECIAL_ATTACK);
            p1->clearActiveHitbox();

            // Practical No. 02: Bresenham spark emission
            Vector2D sparkPos(p2->getPosition().x, p1Hit.offsetY + p1Hit.height * 0.5f);
            particles.emitHitSparks(sparkPos, alu.finalDamage > 20);
            renderer.triggerScreenShake(alu.finalDamage > 20 ? 7.0f : 4.0f, 0.2f);

            // Practical 06 (CO6): Dynamic Combat Point Light flare
            float lR = 1.0f, lG = 0.85f, lB = 0.2f;
            if (p1->getName().find("WARRIOR") != std::string::npos) { lR = 1.0f; lG = 0.45f; lB = 0.1f; }
            else if (p1->getName().find("NINJA") != std::string::npos) { lR = 0.2f; lG = 0.85f; lB = 1.0f; }
            else if (p1->getName().find("BRAWLER") != std::string::npos) { lR = 0.1f; lG = 0.95f; lB = 1.0f; }
            else if (p1->getName().find("PALADIN") != std::string::npos) { lR = 1.0f; lG = 0.95f; lB = 0.35f; }
            else if (p1->getName().find("BERSERKER") != std::string::npos) { lR = 1.0f; lG = 0.15f; lB = 0.1f; }
            else if (p1->getName().find("GLADIATOR") != std::string::npos) { lR = 1.0f; lG = 0.7f; lB = 0.25f; }
            renderer.setCombatLight(sparkPos.x, sparkPos.y, lR, lG, lB, alu.finalDamage > 20 ? 1.5f : 1.0f);
        }

        // 3. Check P2 attacking P1
        Hitbox p2Hit = p2->getActiveHitboxWorld();
        Hitbox p1Hurt = p1->getHurtboxWorld();
        if (p2Hit.active && checkAABB(p2Hit, p1Hurt)) {
            // Practical No. 03: REAL 64-BIT ASSEMBLY DAMAGE CALCULATION!
            CombatALUResult alu = AssemblyDamageEngine::calculateHitDamage(p2->getAttackDamage(), 5, 1);
            latestALUTrace = "P2 HIT! " + alu.assemblyTrace;

            float knockDir = p2->isFacingRightDir() ? 1.0f : -1.0f;
            p1->takeHit((int)alu.finalDamage, knockDir, p2->getState() == FighterState::SPECIAL_ATTACK);
            p2->clearActiveHitbox();

            // Practical No. 02: Bresenham spark emission
            Vector2D sparkPos(p1->getPosition().x, p2Hit.offsetY + p2Hit.height * 0.5f);
            particles.emitHitSparks(sparkPos, alu.finalDamage > 20);
            renderer.triggerScreenShake(alu.finalDamage > 20 ? 7.0f : 4.0f, 0.2f);

            // Practical 06 (CO6): Dynamic Combat Point Light flare
            float lR = 1.0f, lG = 0.85f, lB = 0.2f;
            if (p2->getName().find("WARRIOR") != std::string::npos) { lR = 1.0f; lG = 0.45f; lB = 0.1f; }
            else if (p2->getName().find("NINJA") != std::string::npos) { lR = 0.2f; lG = 0.85f; lB = 1.0f; }
            else if (p2->getName().find("BRAWLER") != std::string::npos) { lR = 0.1f; lG = 0.95f; lB = 1.0f; }
            else if (p2->getName().find("PALADIN") != std::string::npos) { lR = 1.0f; lG = 0.95f; lB = 0.35f; }
